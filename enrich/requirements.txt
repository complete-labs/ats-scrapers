# Dependencies for the repo-only enrichment layer (enrich/).
# Not part of the installable ats-scrapers package — install with:
#   uv pip install -r enrich/requirements.txt
#
# The enrichment layer is a sidecar: it reads the published jobs snapshot
# and writes its own parquet/DuckDB tables. It never imports from
# pipeline/ and never mutates src/, so upstream pulls stay conflict-free.

# Sidecar store and all analytical queries. DuckDB reads the 2.25 GB
# all.parquet lazily and spills to disk, so profiling and joins run in
# bounded memory on a laptop.
duckdb>=1.0
pyarrow>=15.0
# Streaming reads/writes for the batch shard writer. Same version floor
# the upstream publisher uses.
polars>=0.20

# Tier 0 language detection. lingua trades memory for accuracy on short
# text, which is what job titles and truncated descriptions are. The
# low-accuracy mode keeps the model set under ~100 MB.
lingua-language-detector>=2.0

# Offline gazetteer: ISO country -> continent, and city -> lat/lon.
# Ships its own data, so Tier 0 geocoding needs no network and no
# Nominatim rate limit.
geonamescache>=2.0

# Tier 1 / Tier 2 model client. Used for both the sync path and the
# Batch API path; any OpenAI-compatible base_url works via
# ATS_ENRICH_BASE_URL.
openai>=1.40

# Tier 2 agent refetches descriptions through the upstream scrapers,
# which need the [scrapers] extra of the parent package.
